<Only platform="rn">

`QRScanner` is a `View` that draws an overlay on top of whatever you pass as `children`, usually a camera preview. It does not open the camera itself. The overlay dims everything outside a square frame with rounded corners (drawn with `react-native-svg`) and marks the frame with four rounded corner brackets. When `scanLine` is set, a line moves up and down inside the frame with Reanimated while the state is `idle`.

</Only>

<Only platform="rn swift">

The frame size defaults to 66% of the shorter side, capped at 260 points. It is centered horizontally and sits slightly above the vertical center. A `title` sits above the frame and a `description` below it; the flashlight button sits near the bottom edge. The `QRScannerState` type is also exported.

</Only>

```tsx
import { IconProps as PhosphorIconProps } from "phosphor-react-native"
import * as React from "react"
import { View } from "react-native"
import { Path } from "react-native-svg"
import Svg from "react-native-svg"
import { Icon } from "@/components/ui/icon"
import { QRScanner, type QRScannerState } from "@/components/ui/qr-scanner"

const MODULES = 21

const MODULE_SIZE = 6

const CODE_SIZE = MODULES * MODULE_SIZE

const FINDERS = [
  [0, 0],
  [0, 14],
  [14, 0],
]

function isDark(row: number, column: number) {
  for (const [top, left] of FINDERS) {
    const i = row - top
    const j = column - left
    if (i >= 0 && i <= 6 && j >= 0 && j <= 6) {
      return (
        i === 0 ||
        i === 6 ||
        j === 0 ||
        j === 6 ||
        (i >= 2 && i <= 4 && j >= 2 && j <= 4)
      )
    }
    const boxTop = top === 0 ? 0 : top - 1
    const boxLeft = left === 0 ? 0 : left - 1
    if (
      row >= boxTop &&
      row <= boxTop + 7 &&
      column >= boxLeft &&
      column <= boxLeft + 7
    ) {
      return false
    }
  }
  return (row * 7 + column * 11 + row * column) % 5 < 2
}

const CODE_PATH = Array.from({ length: MODULES * MODULES }, (_, index) => {
  const row = Math.floor(index / MODULES)
  const column = index % MODULES
  return isDark(row, column)
    ? `M${column * MODULE_SIZE} ${row * MODULE_SIZE}h${MODULE_SIZE}v${MODULE_SIZE}h-${MODULE_SIZE}Z`
    : ""
}).join("")

function QRCode({ color }: PhosphorIconProps) {
  return (
    <Svg width={CODE_SIZE} height={CODE_SIZE}>
      <Path d={CODE_PATH} fill={color} />
    </Svg>
  )
}

function CameraPlaceholder() {
  return (
    <View className="flex-1 items-center justify-center bg-black">
      <View className="mb-12 -rotate-6 rounded-xl bg-white p-2.5">
        <Icon as={QRCode} className="text-black" />
      </View>
    </View>
  )
}

function Scanner({
  state,
  description,
  scanLine,
}: {
  state?: QRScannerState
  description?: string
  scanLine?: boolean
}) {
  const [torch, setTorch] = React.useState(false)
  return (
    <QRScanner
      className="flex-1"
      state={state}
      scanLine={scanLine}
      title="Scan QR code"
      description={description}
      torch={torch}
      onTorchToggle={setTorch}
    >
      <CameraPlaceholder />
    </QRScanner>
  )
}

export function QrScannerExample() {
  return (
    <Scanner description="Point your camera at a code to pay or connect." />
  )
}
```

```swift
import SwiftUI

struct QRScannerExample: View {
  var body: some View {
    QRScannerDemo(hint: "Point your camera at a code to pay or connect.")
  }
}

struct QRScannerDemo: View {
  var state: QRScannerState = .idle
  let hint: LocalizedStringKey
  var scanLine = false

  @State private var torch = false

  var body: some View {
    QRScanner(
      state: state,
      title: "Scan QR code",
      hint: hint,
      scanLine: scanLine,
      isTorchOn: torch,
      onTorchToggle: { torch.toggle() }
    ) {
      QRScannerCameraPlaceholder()
    }
    .ignoresSafeArea()
  }
}

enum QRCodeSample {
  static let count = 21
  static let module: CGFloat = 6
  static let padding: CGFloat = 10
  static let side = CGFloat(count) * module + padding * 2

  static func isDark(_ r: Int, _ c: Int) -> Bool {
    for (fr, fc) in [(0, 0), (0, 14), (14, 0)] {
      let i = r - fr
      let j = c - fc
      if (0...6).contains(i) && (0...6).contains(j) {
        return i == 0 || i == 6 || j == 0 || j == 6 || ((2...4).contains(i) && (2...4).contains(j))
      }
    }
    if r <= 7 && c <= 7 { return false }
    if r <= 7 && c >= 13 { return false }
    if r >= 13 && c <= 7 { return false }
    return (r * 7 + c * 11 + r * c) % 5 < 2
  }

  static let modules: Path = {
    var path = Path()
    for r in 0..<count {
      for c in 0..<count where isDark(r, c) {
        path.addRect(CGRect(x: padding + CGFloat(c) * module, y: padding + CGFloat(r) * module, width: module, height: module))
      }
    }
    return path
  }()
}

struct QRScannerCameraPlaceholder: View {
  var body: some View {
    GeometryReader { proxy in
      Canvas { context, _ in
        let card = CGRect(origin: .zero, size: CGSize(width: QRCodeSample.side, height: QRCodeSample.side))
        context.fill(Path(roundedRect: card, cornerRadius: Radius.xl, style: .continuous), with: .color(Palette.white))
        context.fill(QRCodeSample.modules, with: .color(Palette.black))
      }
      .frame(width: QRCodeSample.side, height: QRCodeSample.side)
      .rotationEffect(.degrees(-6))
      .position(x: proxy.size.width / 2, y: proxy.size.height / 2 - 24)
    }
    .background(Palette.black)
    .accessibilityHidden(true)
  }
}
```

```kotlin
import androidx.compose.ui.geometry.Rect
import androidx.compose.ui.graphics.Path
import androidx.compose.ui.unit.dp
import com.esuiss.ui.components.QrScanner
import com.esuiss.ui.components.QrScannerState
import com.esuiss.ui.theme.DarkColors
import com.esuiss.ui.theme.Palette

@Composable
fun QRScannerExample() {
    QrScannerDemo(hint = "Point your camera at a code to pay or connect.")
}

const val QrModules = 21

val QrModule = 6.dp

fun qrModuleDark(r: Int, c: Int): Boolean {
    val boxes = listOf(0 to 0, 0 to 13, 13 to 0)
    for ((top, left) in boxes) {
        if (r in top until top + 8 && c in left until left + 8) {
            val fr = if (top == 0) 0 else 14
            val fc = if (left == 0) 0 else 14
            val i = r - fr
            val j = c - fc
            if (i !in 0..6 || j !in 0..6) return false
            return i == 0 || i == 6 || j == 0 || j == 6 || (i in 2..4 && j in 2..4)
        }
    }
    return (r * 7 + c * 11 + r * c) % 5 < 2
}

@Composable
fun BoxScope.QrScannerCameraPlaceholder() {
    val colors = DarkColors
    val modules = remember {
        (0 until QrModules).flatMap { r -> (0 until QrModules).filter { c -> qrModuleDark(r, c) }.map { c -> r to c } }
    }
    Box(
        modifier = Modifier.matchParentSize().background(Palette.black),
        contentAlignment = Alignment.Center,
    ) {
        Box(
            Modifier
                .offset(y = (-24).dp)
                .rotate(-6f)
                .background(Palette.white, RoundedCornerShape(12.dp))
                .padding(10.dp),
        ) {
            Canvas(Modifier.size(QrModule * QrModules)) {
                val unit = size.width / QrModules
                val path = Path().apply {
                    modules.forEach { (r, c) -> addRect(Rect(c * unit, r * unit, (c + 1) * unit, (r + 1) * unit)) }
                }
                drawPath(path, Palette.black)
            }
        }
    }
}

@Composable
fun QrScannerDemo(
    state: QrScannerState = QrScannerState.Idle,
    hint: String? = null,
    scanLine: Boolean = false,
) {
    var torch by rememberSaveable { mutableStateOf(false) }
    QrScanner(
        modifier = Modifier.fillMaxSize(),
        state = state,
        title = "Scan QR code",
        hint = hint,
        scanLine = scanLine,
        torch = torch,
        onTorchToggle = { torch = it },
    ) {
        QrScannerCameraPlaceholder()
    }
}
```

## Accessibility

- The dimmed surround and brackets are hidden from VoiceOver and TalkBack.
- The overlay is a polite live region, so changes to the title and description are announced on Android.
- The flashlight button has the `button` role, uses `torchLabel` as its label and reports `torch` as its selected state.
- State changes to `success` and `error` play a matching haptic notification when a haptics adapter is provided through `HapticsContext`.
- The shake and scan-line animations respect the system reduce motion setting, and the success scale is skipped when reduce motion is on.

</Only>

## API Reference

<Only platform="rn">

### QRScanner

The camera overlay with the scan frame, its labels and the flashlight button.

| Prop | Type | Default |
| --- | --- | --- |
| `state` | `QRScannerState` | `"idle"` |
| `title` | `React.ReactNode` | - |
| `description` | `React.ReactNode` | - |
| `scanLine` | `boolean` | `false` |
| `torch` | `boolean` | `false` |
| `onTorchToggle` | `(torch: boolean) => void` | - |
| `torchLabel` | `string` | `"Flashlight"` |
| `frameSize` | `number` | - |

Also accepts the props of `React.ComponentProps<typeof View>`.

Render your camera view as `children` and size the scanner with `className`. When there are no children, the background falls back to `bg-black`.

`state` sets the bracket color and triggers haptics, the success scale and the error shake. The scan line restarts whenever the state returns to `idle`. `title` and `description` accept strings, which are wrapped in a `Text`, or your own elements. `frameSize` is the side length of the scan frame in points and overrides the automatic size.

Pass `onTorchToggle` together with a controlled `torch` value to show the flashlight button; it is called with the next torch value when the button is pressed. When `torch` is on, the button fills white with a black icon in the fill weight. `torchLabel` is the button's accessibility label.

</Only>
