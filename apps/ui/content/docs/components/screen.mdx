<Only platform="rn">

`Screen` is a `View` that fills the screen, paints the `bg-background` color (or `bg-muted` for the `secondary` variant) and adds padding for the safe area insets on the edges you choose. It also renders a `StatusBar` with light or dark content to match the color scheme.

`ScreenContent` lays out the body with padding and gaps. With `scrollable`, it renders a `KeyboardScrollView` so content scrolls and stays above the keyboard.

</Only>

## Usage

<Only platform="rn">

<OnlyStyling styling="uniwind nativewind">

```tsx
import { View } from "react-native"
import { Screen, ScreenContent } from "@/components/ui/screen"
import { Text } from "@/components/ui/text"

function Rows({ count = 4 }: { count?: number }) {
  return (
    <View className="overflow-hidden rounded-2xl bg-muted">
      {Array.from({ length: count }, (_, index) => `Row ${index + 1}`).map(
        (label, index) => (
          <View
            key={label}
            className={
              index > 0 ? "border-t border-border px-4 py-3" : "px-4 py-3"
            }
          >
            <Text>{label}</Text>
          </View>
        )
      )}
    </View>
  )
}

export function ScreenExample() {
  return (
    <Screen className="-m-4">
      <ScreenContent>
        <Text variant="h2">Account</Text>
        <Text variant="muted">
          The screen pads every safe-area edge and sets the status bar style
          from the color scheme.
        </Text>
        <Rows />
      </ScreenContent>
    </Screen>
  )
}
```

<Only platform="swift">

```swift
import SwiftUI

struct ScreenExample: View {
  var body: some View {
    Screen {
      ScreenContent(scroll: false) {
        Text("Account").textVariant(.h2)
        Text("The screen pads every safe-area edge and sets the status bar style from the color scheme.")
          .textVariant(.muted)
        ScreenRows()
      }
    }
  }
}

struct ScreenRows: View {
  var labels: [String] = (1...4).map { "Row \($0)" }
  var background: Color = EsuissColor.muted

  var body: some View {
    VStack(spacing: 0) {
      ForEach(Array(labels.enumerated()), id: \.offset) { index, label in
        Text(verbatim: label)
          .textVariant(.default)
          .frame(maxWidth: .infinity, alignment: .leading)
          .padding(.horizontal, 16)
          .padding(.vertical, 12)
          .overlay(alignment: .top) {
            if index > 0 {
              EsuissColor.border.frame(height: 1)
            }
          }
      }
    }
    .background(background)
    .clipShape(RoundedRectangle(cornerRadius: Radius.xl2, style: .continuous))
  }
}
```

<Only platform="kotlin">

```kotlin
import androidx.compose.foundation.background
import androidx.compose.foundation.layout.Column
import androidx.compose.foundation.layout.fillMaxWidth
import androidx.compose.foundation.layout.padding
import androidx.compose.foundation.shape.RoundedCornerShape
import androidx.compose.runtime.Composable
import androidx.compose.ui.Modifier
import androidx.compose.ui.draw.clip
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.unit.dp
import com.esuiss.ui.components.Screen
import com.esuiss.ui.components.ScreenContent
import com.esuiss.ui.components.Separator
import com.esuiss.ui.components.Text
import com.esuiss.ui.components.TextVariant
import com.esuiss.ui.theme.EsuissTheme
import com.esuiss.ui.theme.Radius

@Composable
fun ScreenRows(labels: List<String>, background: Color) {
    Column(
        Modifier
            .fillMaxWidth()
            .clip(RoundedCornerShape(Radius.xl2))
            .background(background),
    ) {
        labels.forEachIndexed { index, label ->
            if (index > 0) Separator()
            Text(label, Modifier.fillMaxWidth().padding(horizontal = 16.dp, vertical = 12.dp))
        }
    }
}

fun screenRowLabels(count: Int): List<String> = (1..count).map { "Row $it" }

@Composable
fun ScreenDefault() {
    Screen {
        ScreenContent {
            Text("Account", variant = TextVariant.H2)
            Text(
                "The screen pads every safe-area edge and sets the status bar style from the color scheme.",
                variant = TextVariant.Muted,
            )
            ScreenRows(screenRowLabels(4), EsuissTheme.colors.muted)
        }
    }
}
```

</Only>

<Only platform="rn">

Wrap each screen in `Screen` and put the body in `ScreenContent`. Use `variant="secondary"` for grouped screens such as settings.

Pass `edges` to pad only some sides, for example `["top"]` when a tab bar or bottom bar handles the bottom inset. Set `scrollable` on `ScreenContent` for content taller than the screen or for forms.

The app must be wrapped in a `SafeAreaProvider` from `react-native-safe-area-context`. `ScreenContext` exposes the active `edges` to children.

</Only>

## API Reference

<Only platform="rn">

<ApiPart index={1}>

### Screen

The root view that pads for safe area insets.

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `edges` | `readonly ScreenEdge[]` | `ALL_EDGES` | Which sides get safe area padding: any of `top`, `right`, `bottom`, `left`. Defaults to all four. |
| `variant` | `"default" \| "secondary"` | `"default"` | Background: `default` or `secondary`. |

Also accepts the props of `ViewProps`.

</ApiPart>

<ApiPart index={2}>

### ScreenContent

The padded body, optionally scrollable.

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `scrollable` | `boolean` | `false` | Renders a `KeyboardScrollView` instead of a `View`. Defaults to `false`. |

Also accepts the props of `ViewProps`.

</ApiPart>

</Only>

## Notes

- Uses the `bg-background` and `bg-muted` tokens.
- The status bar uses `light-content` in dark mode and `dark-content` in light mode, following the system appearance.
- Requires `react-native-safe-area-context` and the Keyboard Scroll View component.
