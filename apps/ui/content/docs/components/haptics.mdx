<Only platform="rn">

Components such as Amount Input, Chart and Context Menu play haptic feedback through `useHaptics()`. By default nothing happens: `HapticsProvider` lets you plug in an adapter that calls the haptics library of your choice, so the components stay free of a native dependency.

</Only>

## Usage

<Only platform="rn">

<OnlyStyling styling="uniwind nativewind">

```tsx
import * as React from "react"
import { View } from "react-native"
import { Stepper, StepperDecrement, StepperGroup, StepperIncrement, StepperInput, StepperSeparator } from "@/components/interactions/stepper"
import { Checkbox } from "@/components/ui/checkbox"
import { Chip } from "@/components/ui/chip"
import { type HapticsAdapter, HapticsProvider } from "@/components/ui/haptics"
import { Slider } from "@/components/ui/slider"
import { Switch } from "@/components/ui/switch"
import { Text } from "@/components/ui/text"

function useLoggingAdapter() {
  const [log, setLog] = React.useState({ last: "None", count: 0 })
  const adapter = React.useMemo<HapticsAdapter>(() => {
    const record = (event: string) =>
      setLog((current) => ({ last: event, count: current.count + 1 }))
    return {
      impact: (style) => record(`impact ${style}`),
      selection: () => record("selection"),
      notification: (type) => record(`notification ${type}`),
    }
  }, [])
  return { adapter, log }
}

function HapticLog({ last, count }: { last: string; count: number }) {
  return (
    <View className="flex-row items-center justify-between rounded-xl bg-muted px-4 py-3">
      <Text className="text-sm text-muted-foreground">Last haptic</Text>
      <Text
        role="status"
        accessibilityLiveRegion="polite"
        className="text-sm font-medium"
      >
        {count > 0 ? `${last} (${count})` : last}
      </Text>
    </View>
  )
}

function Row({
  label,
  children,
}: {
  label: string
  children: React.ReactNode
}) {
  return (
    <View className="min-h-11 flex-row items-center justify-between gap-4">
      <Text>{label}</Text>
      {children}
    </View>
  )
}

function Controls() {
  const [wifi, setWifi] = React.useState(true)
  const [terms, setTerms] = React.useState(false)
  const [volume, setVolume] = React.useState(40)
  const [copies, setCopies] = React.useState(1)

  return (
    <View className="w-full gap-2">
      <Row label="Wi-Fi">
        <Switch
          accessibilityLabel="Wi-Fi"
          checked={wifi}
          onCheckedChange={setWifi}
        />
      </Row>
      <Row label="Accept terms">
        <Checkbox
          accessibilityLabel="Accept terms"
          checked={terms}
          onCheckedChange={setTerms}
        />
      </Row>
      <Row label="Copies">
        <Stepper value={copies} onValueChange={setCopies} min={1} max={9}>
          <StepperInput accessibilityLabel="Copies" />
          <StepperGroup>
            <StepperDecrement />
            <StepperSeparator />
            <StepperIncrement />
          </StepperGroup>
        </Stepper>
      </Row>
      <View className="flex-row gap-2 py-2">
        <Chip>Open now</Chip>
        <Chip>Top rated</Chip>
      </View>
      <Slider
        value={volume}
        step={10}
        onValueChange={(next) => setVolume(next)}
        accessibilityLabel="Volume"
      />
    </View>
  )
}

function ControlsDemo({ enabled = true }: { enabled?: boolean }) {
  const { adapter, log } = useLoggingAdapter()
  return (
    <HapticsProvider adapter={adapter} enabled={enabled}>
      <View className="w-full gap-4">
        <HapticLog {...log} />
        <Controls />
      </View>
    </HapticsProvider>
  )
}

export function HapticsExample() {
  return (
    <ControlsDemo />
  )
}
```

import SwiftUI

struct HapticsRow<Content: View>: View {
  let label: LocalizedStringKey
  @ViewBuilder let content: () -> Content

  var body: some View {
    HStack(spacing: 16) {
      Text(label)
        .textStyle(.base)
        .foregroundStyle(EsuissColor.foreground)
      Spacer(minLength: 0)
      content()
    }
    .frame(minHeight: 44)
  }
}

struct HapticsControls: View {
  @State private var wifi = true
  @State private var terms = false
  @State private var volume = 40.0
  @State private var copies = 1
  @State private var openNow = false
  @State private var topRated = false

  var body: some View {
    VStack(alignment: .leading, spacing: 8) {
      HapticsRow(label: "Wi-Fi") {
        Toggle(isOn: $wifi) { EmptyView() }
          .toggleStyle(.switch(size: .default))
          .accessibilityLabel(Text("Wi-Fi"))
      }
      HapticsRow(label: "Accept terms") {
        Toggle(isOn: $terms) { EmptyView() }
          .toggleStyle(.checkbox)
          .accessibilityLabel(Text("Accept terms"))
      }
      HapticsRow(label: "Copies") {
        ValueStepper(value: $copies, in: 1...9) {
          StepperInput()
            .accessibilityLabel(Text("Copies"))
          StepperGroup {
            StepperDecrement()
            StepperSeparator()
            StepperIncrement()
          }
        }
      }
      HStack(spacing: 8) {
        Chip("Open now", isSelected: $openNow)
        Chip("Top rated", isSelected: $topRated)
      }
      .padding(.vertical, 8)
      ValueSlider(value: $volume, in: 0...100, step: 10)
        .accessibilityLabel(Text("Volume"))
    }
    .frame(maxWidth: .infinity, alignment: .leading)
  }
}

@Composable
fun HapticsExample() {
    HapticsControlsDemo(enabled = true)
}

class HapticsLog(private val system: HapticFeedback) : HapticFeedback {
    var last by mutableStateOf("None")
    var count by mutableIntStateOf(0)

    override fun performHapticFeedback(hapticFeedbackType: HapticFeedbackType) {
        last = hapticFeedbackType.toString()
        count += 1
        system.performHapticFeedback(hapticFeedbackType)
    }
}

@Composable
fun HapticsLogging(content: @Composable (HapticsLog) -> Unit) {
    val system = LocalHapticFeedback.current
    val log = remember(system) { HapticsLog(system) }
    CompositionLocalProvider(LocalHapticFeedback provides log) {
        content(log)
    }
}

@Composable
fun HapticsLogRow(log: HapticsLog) {
    val colors = EsuissTheme.colors
    Row(
        modifier = Modifier
            .fillMaxWidth()
            .background(colors.muted, RoundedCornerShape(Radius.xl))
            .padding(horizontal = 16.dp, vertical = 12.dp),
        horizontalArrangement = Arrangement.SpaceBetween,
        verticalAlignment = Alignment.CenterVertically,
    ) {
        Text("Last haptic", style = Typography.sm.copy(color = colors.mutedForeground))
        Text(
            if (log.count > 0) "${log.last} (${log.count})" else log.last,
            modifier = Modifier.semantics { liveRegion = LiveRegionMode.Polite },
            style = Typography.sm.copy(fontWeight = FontWeight.Medium),
        )
    }
}

@Composable
fun HapticsRow(label: String, content: @Composable () -> Unit) {
    Row(
        modifier = Modifier.fillMaxWidth().heightIn(min = 44.dp),
        horizontalArrangement = Arrangement.spacedBy(16.dp),
        verticalAlignment = Alignment.CenterVertically,
    ) {
        Text(label, modifier = Modifier.weight(1f))
        content()
    }
}

@Composable
fun HapticsControls() {
    var wifi by rememberSaveable { mutableStateOf(true) }
    var terms by rememberSaveable { mutableStateOf(false) }
    var volume by rememberSaveable { mutableFloatStateOf(40f) }
    var copies by rememberSaveable { mutableDoubleStateOf(1.0) }
    var openNow by rememberSaveable { mutableStateOf(false) }
    var topRated by rememberSaveable { mutableStateOf(false) }
    Column(Modifier.fillMaxWidth(), verticalArrangement = Arrangement.spacedBy(8.dp)) {
        HapticsRow("Wi-Fi") {
            Switch(wifi, { wifi = it }, Modifier.semantics { contentDescription = "Wi-Fi" })
        }
        HapticsRow("Accept terms") {
            Checkbox(terms, { terms = it }, Modifier.semantics { contentDescription = "Accept terms" })
        }
        HapticsRow("Copies") {
            Stepper(copies, { copies = it }, min = 1.0, max = 9.0) {
                StepperInput(contentDescription = "Copies")
                StepperGroup {
                    StepperDecrement()
                    StepperSeparator()
                    StepperIncrement()
                }
            }
        }
        Row(Modifier.padding(vertical = 8.dp), horizontalArrangement = Arrangement.spacedBy(8.dp)) {
            Chip("Open now", openNow, { openNow = it })
            Chip("Top rated", topRated, { topRated = it })
        }
        Slider(volume, { volume = it }, step = 10f, contentDescription = "Volume")
    }
}

        }
    }
}

## API Reference

<Only platform="rn">

### HapticsProvider

Supplies the adapter to every component below it.

| Prop | Type | Default |
| --- | --- | --- |
| `adapter` | `HapticsAdapter \| null` | - |
| `enabled` | `boolean` | `true` |
| `children` | `React.ReactNode` | - |

Wrap your app in `HapticsProvider` and pass an `adapter` object that calls your haptics library. The adapter has three methods: `impact(style)` with `light`, `medium`, `heavy`, `soft` or `rigid`; `selection()`; and `notification(type)` with `success`, `warning` or `error`. The types `HapticsAdapter`, `HapticsImpactStyle` and `HapticsNotificationType` are exported for typing your adapter.

Call `useHaptics()` in your own components to get the same methods. Without a provider, or when disabled, they do nothing. Haptics have no visual output; pair important feedback with a visible change.

</Only>
